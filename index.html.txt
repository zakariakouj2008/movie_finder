  <script>

    async function sendImage() {

      const fileInput = document.getElementById("image");
      const status = document.getElementById("status");

      const file = fileInput.files[0];

      if (!file) {
        status.innerText = "❌ اختار صورة أولاً";
        return;
      }

      status.innerText = "⏳ كنصيفط الصورة...";

      const formData = new FormData();

      formData.append("image", file);

      const webhook =
        "https://hook.eu1.make.com/fyc5rlniw3axeere56oxckgne8b817g1";

      try {

        const response = await fetch(webhook, {
          method: "POST",
          body: formData
        });

        if (response.ok) {

          status.innerText =
            "✅ الصورة تصيفطات لـ Make بنجاح!";

        } else {

          status.innerText =
            "❌ وقع مشكل فإرسال الصورة.";

        }

      } catch (error) {

        console.error(error);

        status.innerText =
          "❌ ما قدرناش نتاصلو بـ Make.";

      }

    }

  </script>
